function doPost(e) {
  try {
    const data = getRequestData_(e);
    const action = clean_(data.action);

    if (action === "uploadPaidMaster") {
      return uploadPaidMaster_(data);
    }

    if (action === "submitTD") {
      return submitLineTd_(data);
    }

    return submitRevenuePayment_(data);
  } catch (error) {
    return jsonResponse_({
      status: "error",
      message: error && error.message ? error.message : "Unknown error"
    });
  }
}

function doGet(e) {
  try {
    const action = clean_(e && e.parameter && e.parameter.action);

    if (action === "getEntries") {
      return getRevenueEntries_();
    }

    if (action === "checkPaid") {
      return checkPaidEntry_(e.parameter || {});
    }

    if (action === "checkUploadedPaid") {
      return checkUploadedPaid_(e.parameter || {});
    }

    if (action === "getUploadedPaidEntries") {
      return getUploadedPaidEntries_(e.parameter || {});
    }

    if (action === "checkTD") {
      return checkLineTd_(e.parameter || {});
    }

    if (action === "getTDEntries") {
      return getLineTdEntries_(e.parameter || {});
    }

    return jsonResponse_({
      status: "success",
      message: "Revenue Collection Script Live Hai"
    });
  } catch (error) {
    return jsonResponse_({
      status: "error",
      message: error && error.message ? error.message : "Unknown error"
    });
  }
}

function submitRevenuePayment_(data) {
  const ss = getSpreadsheet_();
  const sheet = getOrCreateSheet_(ss, COLLECTION_SHEET_NAME, COLLECTION_HEADERS);

  const dcName = clean_(data.dc_name);
  const ivrsNo = normalizeIvrs_(data.ivrs_no);
  const amountPaid = clean_(data.amount_paid);

  if (!ivrsNo) throw new Error("IVRS No missing hai");
  if (!amountPaid) throw new Error("Amount Paid missing hai");

  const existingPayments = findCollectionPayments_(sheet, ivrsNo, dcName);
  if (existingPayments.length >= 2) {
    throw new Error("Already submitted 2 times");
  }

  const now = new Date();
  const date = Utilities.formatDate(now, Session.getScriptTimeZone() || "Asia/Kolkata", "dd/MM/yyyy");
  const time = Utilities.formatDate(now, Session.getScriptTimeZone() || "Asia/Kolkata", "HH:mm");

  const row = [[
    dcName,
    ivrsNo,
    clean_(data.consumer_name),
    clean_(data.father_name),
    clean_(data.village),
    clean_(data.hq_name),
    clean_(data.tarrif_category),
    clean_(data.mobile_no),
    clean_(data.arrears),
    clean_(data.net_bill),
    amountPaid,
    date,
    time
  ]];

  const startRow = Math.max(sheet.getLastRow() + 1, 2);
  const range = sheet.getRange(startRow, 1, 1, COLLECTION_HEADERS.length);
  range.setValues(row);
  styleDataRange_(range);

  updateLineTdPaidStatus_(ss, dcName, ivrsNo, amountPaid, date, time);

  return jsonResponse_({
    status: "success",
    message: "Revenue paid amount submit ho gaya",
    payment_count: existingPayments.length + 1
  });
}

function uploadPaidMaster_(data) {
  const ss = getSpreadsheet_();
  const sheet = getOrCreateSheet_(ss, PAID_MASTER_SHEET_NAME, PAID_MASTER_HEADERS);

  const dcName = clean_(data.dc_name);
  const entries = parseEntries_(data.entries_json);

  if (!dcName) throw new Error("DC Name missing hai");
  if (!entries.length) throw new Error("Paid entries missing hai");

  const now = new Date();
  const uploadDate = Utilities.formatDate(now, Session.getScriptTimeZone() || "Asia/Kolkata", "dd/MM/yyyy");
  const uploadTime = Utilities.formatDate(now, Session.getScriptTimeZone() || "Asia/Kolkata", "HH:mm");

  replacePaidMasterDcRows_(sheet, dcName);

  const rows = entries.map(function(entry) {
    const ivrsNo = normalizeIvrs_(entry.ivrs_no || entry.ivrsNo);
    if (!ivrsNo) return null;

    return [
      dcName,
      ivrsNo,
      clean_(entry.amount_paid || entry.amountPaid),
      clean_(entry.payment_date || entry.paymentDate),
      clean_(entry.payment_count || entry.paymentCount),
      clean_(entry.source_type || entry.sourceType),
      clean_(entry.source),
      clean_(entry.pay_mode || entry.payMode),
      uploadDate,
      uploadTime
    ];
  }).filter(Boolean);

  if (!rows.length) throw new Error("Valid IVRS No missing hai");

  const startRow = Math.max(sheet.getLastRow() + 1, 2);
  const range = sheet.getRange(startRow, 1, rows.length, PAID_MASTER_HEADERS.length);
  range.setValues(rows);
  styleDataRange_(range);

  rows.forEach(function(row) {
    updateLineTdPaidStatus_(ss, row[0], row[1], row[2], row[3], "");
  });

  return jsonResponse_({
    status: "success",
    message: "Paid master upload ho gaya",
    rows_saved: rows.length
  });
}

function checkUploadedPaid_(params) {
  const ss = getSpreadsheet_();
  const sheet = getOrCreateSheet_(ss, PAID_MASTER_SHEET_NAME, PAID_MASTER_HEADERS);

  const dcName = clean_(params.dc_name);
  const ivrsNo = normalizeIvrs_(params.ivrs_no);

  if (!ivrsNo) throw new Error("IVRS No missing hai");

  const lastRow = sheet.getLastRow();
  if (lastRow < 2) {
    return jsonResponse_({ status: "success", paid: false });
  }

  const values = sheet.getRange(2, 1, lastRow - 1, PAID_MASTER_HEADERS.length).getValues();

  for (let i = values.length - 1; i >= 0; i--) {
    const row = values[i];
    if (normalizeIvrs_(row[1]) === ivrsNo && (!dcName || clean_(row[0]) === dcName)) {
      return jsonResponse_({
        status: "success",
        paid: true,
        data: {
          dc_name: row[0],
          ivrs_no: row[1],
          amount_paid: row[2],
          payment_date: row[3],
          payment_count: row[4],
          source_type: row[5],
          source: row[6],
          pay_mode: row[7]
        }
      });
    }
  }

  return jsonResponse_({ status: "success", paid: false });
}

function getPaidStatusText_(ss, dcName, ivrsNo) {
  const collectionSheet = getOrCreateSheet_(ss, COLLECTION_SHEET_NAME, COLLECTION_HEADERS);
  const paidSheet = getOrCreateSheet_(ss, PAID_MASTER_SHEET_NAME, PAID_MASTER_HEADERS);

  const manualPayments = findCollectionPayments_(collectionSheet, ivrsNo, dcName);
  if (manualPayments.length) {
    const last = manualPayments[manualPayments.length - 1];
    return "Paid By Staff: Rs " + clean_(last.amount_paid) + " | Date: " + clean_(last.date) + " | Time: " + formatTime_(last.time);
  }

  const uploaded = findUploadedPaid_(paidSheet, ivrsNo, dcName);
  if (uploaded) {
    return "NGB Cashlist: Rs " + clean_(uploaded.amount_paid) + " | Date: " + clean_(uploaded.payment_date);
  }

  return "";
}

function findUploadedPaid_(sheet, ivrsNo, dcName) {
  const targetIvrs = normalizeIvrs_(ivrsNo);

  const values = sheet.getRange(2, 1, lastRow - 1, PAID_MASTER_HEADERS.length).getValues();

        payment_count: row[4],
        source_type: row[5]

}

function updateLineTdPaidStatus_(ss, dcName, ivrsNo, amountPaid, date, time) {
  const sheet = getOrCreateSheet_(ss, LINE_TD_SHEET_NAME, LINE_TD_HEADERS);
  const targetDc = clean_(dcName);
  const targetIvrs = normalizeIvrs_(ivrsNo);
  const lastRow = sheet.getLastRow();

  if (!targetDc || !targetIvrs || lastRow < 2) return;

  const values = sheet.getRange(2, 1, lastRow - 1, LINE_TD_HEADERS.length).getValues();
  const status = time
    ? "Paid: Rs " + clean_(amountPaid) + " | Date: " + clean_(date) + " | Time: " + formatTime_(time)
    : "NGB Cashlist: Rs " + clean_(amountPaid) + " | Date: " + clean_(date);

  values.forEach(function(row, index) {
    if (clean_(row[0]) === targetDc && normalizeIvrs_(row[1]) === targetIvrs) {
      sheet.getRange(index + 2, LINE_TD_HEADERS.length).setValue(status);
    }
  });
}

function repairLineTdPaidStatusTimes() {
  const ss = getSpreadsheet_();
  const sheet = getOrCreateSheet_(ss, LINE_TD_SHEET_NAME, LINE_TD_HEADERS);
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  const statusRange = sheet.getRange(2, LINE_TD_HEADERS.length, lastRow - 1, 1);
  const values = statusRange.getDisplayValues();
  let fixedCount = 0;

  const fixedValues = values.map(function(row) {
    const status = clean_(row[0]);
    if (!status || status.indexOf("Time:") === -1) return [status];

    const prefix = status.split("Time:")[0] + "Time: ";
    const timePart = status.substring(status.indexOf("Time:") + 5);
    const match = timePart.match(/(\d{1,2}):(\d{2})(?::\d{2})?/);
    if (!match) return [status];

    const shortTime = String(match[1]).padStart(2, "0") + ":" + match[2];
    const corrected = prefix + shortTime;
    if (corrected !== status) fixedCount += 1;
    return [corrected];
  });

  statusRange.setValues(fixedValues);
  styleDataRange_(statusRange);
  SpreadsheetApp.getActive().toast(fixedCount + " PAID STATUS time fixed", "LINE TD REPORT");
}

function saveTdPhotoIfProvided_(data) {
  const base64 = clean_(data.photo_base64);
  const fileName = clean_(data.photo_name) || ("line-td-photo-" + Date.now() + ".jpg");
  const mimeType = clean_(data.photo_mime_type) || "image/jpeg";

  if (!base64) {
    return { link: "", blob: null };
  }

  const pureBase64 = base64.indexOf(",") > -1 ? base64.split(",")[1] : base64;
  const bytes = Utilities.base64Decode(pureBase64);
  const blob = Utilities.newBlob(bytes, mimeType, fileName);
  const file = DriveApp.createFile(blob);

  try {
    file.setSharing(DriveApp.Access.ANYONE_WITH_LINK, DriveApp.Permission.VIEW);
  } catch (_) {}

  return {
    link: file.getUrl(),
    blob: blob
  };
}

function replacePaidMasterDcRows_(sheet, dcName) {
  const lastRow = sheet.getLastRow();
  if (lastRow < 2) return;

  const values = sheet.getRange(2, 1, lastRow - 1, PAID_MASTER_HEADERS.length).getValues();
  const keptRows = values.filter(function(row) {
    return clean_(row[0]) !== dcName;
  });

  sheet.getRange(2, 1, lastRow - 1, PAID_MASTER_HEADERS.length).clearContent();

  if (keptRows.length) {
    const range = sheet.getRange(2, 1, keptRows.length, PAID_MASTER_HEADERS.length);
    range.setValues(keptRows);
    styleDataRange_(range);
  }
}

function getOrCreateSheet_(ss, sheetName, headers) {
  if (!ss) throw new Error("Spreadsheet open nahi ho rahi");

  let sheet = ss.getSheetByName(sheetName);
  if (!sheet) sheet = ss.insertSheet(sheetName);

  setupSheetLayout_(sheet, headers);
  return sheet;
}

function setupSheetLayout_(sheet, headers) {
  const headerRange = sheet.getRange(1, 1, 1, headers.length);
  headerRange.setValues([headers]);
  headerRange
    .setFontWeight("bold")
    .setBackground("#bfdbfe")
    .setFontColor("#1e3a8a")
    .setHorizontalAlignment("center")
    .setVerticalAlignment("middle")
    .setBorder(true, true, true, true, true, true);

  sheet.setFrozenRows(1);

  if (!sheet.getFilter()) {
    sheet.getRange(1, 1, Math.max(sheet.getLastRow(), 1), headers.length).createFilter();
  }

  protectHeader_(sheet, headers.length);
}

function protectHeader_(sheet, totalCols) {
  try {
    const headerRange = sheet.getRange(1, 1, 1, totalCols);
    const existing = sheet.getProtections(SpreadsheetApp.ProtectionType.RANGE).some(function(protection) {
      return protection.getRange().getA1Notation() === headerRange.getA1Notation();
    });

    if (!existing) {
      const protection = headerRange.protect();
      protection.setDescription(sheet.getName() + " header locked");
      protection.removeEditors(protection.getEditors());
      if (protection.canDomainEdit()) protection.setDomainEdit(false);
    }
  } catch (_) {}
}

function styleDataRange_(range) {
  range
    .setHorizontalAlignment("center")
    .setVerticalAlignment("middle")
    .setBorder(true, true, true, true, true, true);
}

function getRequestData_(e) {
  if (!e) return {};

  if (e.postData && e.postData.contents) {
    const raw = e.postData.contents;
    const type = String(e.postData.type || "").toLowerCase();

    if (type.indexOf("application/json") > -1 || raw.trim().charAt(0) === "{") {
      return JSON.parse(raw);
    }
  }

  return e.parameter || {};
}

function parseEntries_(value) {
  if (Array.isArray(value)) return value;

  if (typeof value === "string" && value.trim()) {
    return JSON.parse(value);
  }

  return [];
}

function normalizeIvrs_(value) {
  return String(value || "").replace(/[^0-9]/g, "").trim();
}

function formatTime_(value) {
  const tz = Session.getScriptTimeZone() || "Asia/Kolkata";

  if (Object.prototype.toString.call(value) === "[object Date]" && !isNaN(value.getTime())) {
    return Utilities.formatDate(value, tz, "HH:mm");
  }

  const raw = clean_(value);
  const match = raw.match(/(\d{1,2}):(\d{2})(?::\d{2})?/);
  if (match) {
    return String(match[1]).padStart(2, "0") + ":" + match[2];
  }

  return raw;
}

function clean_(value) {
  return String(value || "").replace(/\s+/g, " ").trim();
}

function jsonResponse_(obj) {
  return ContentService
    .createTextOutput(JSON.stringify(obj))
    .setMimeType(ContentService.MimeType.JSON);
}
